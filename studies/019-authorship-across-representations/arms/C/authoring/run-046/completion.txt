package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	evidence := object.get(input, "evidence", {})
	not "financial-evidence" in evidence
} else := result if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "present"
	result := unreadable_result
}

unreadable_result := result if {
	count(candidate_outcomes) == 1
	some result in candidate_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

candidate_outcomes := {result |
	vendor := object.get(input, "vendor", {})
	some risk in possible_risks(vendor)
	some spend in possible_spends(vendor)
	some country in possible_countries(vendor)
	result := readable_outcome(vendor, risk, spend, country)
}

possible_risks(vendor) := [vendor.riskScore] if {
	"riskScore" in vendor
} else := [0, 39, 40, 69, 70, 89, 90, 100]

possible_spends(vendor) := [vendor.requestedSpend] if {
	"requestedSpend" in vendor
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

possible_countries(vendor) := [vendor.countryRisk] if {
	"countryRisk" in vendor
} else := ["LOW", "MEDIUM", "HIGH"]

readable_outcome(vendor, _, _, _) := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	vendor.sanctionsStatus == "UNKNOWN"
} else := {"disposition": "unresolved", "reasons": ["exception-escalation"]} if {
	vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	evidence := object.get(input, "evidence", {})
	evidence["insurance-certificate"] == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	evidence := object.get(input, "evidence", {})
	evidence["insurance-certificate"] == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	evidence := object.get(input, "evidence", {})
	not "insurance-certificate" in evidence
} else := {"disposition": "approve", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"

	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []} if {
	vendor.sanctionsStatus == "CLEAR"
