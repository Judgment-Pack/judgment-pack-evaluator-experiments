	"unreported",

	"insurance-certificate",
	"unreported",
)

vendor := object.get(input, "vendor", {})

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_status == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_status == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	vendor.sanctionsStatus == "CLEAR"
	result := numeric_result
}

risk_candidates := [vendor.riskScore] if {
	"riskScore" in object.keys(vendor)
} else := numbers.range(0, 100)

spend_candidates := [vendor.requestedSpend] if {
	"requestedSpend" in object.keys(vendor)
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

country_candidates := [vendor.countryRisk] if {
	"countryRisk" in object.keys(vendor)
} else := ["LOW", "MEDIUM", "HIGH"]

possible_outcomes contains outcome if {
	some risk in risk_candidates
	some spend in spend_candidates
	some country in country_candidates
	outcome := readable_outcome(risk, spend, country)
}

numeric_result := outcome if {
	count(possible_outcomes) == 1
	some outcome in possible_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

readable_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000

} else := {"disposition": "enhanced-review", "reasons": []} if {
	country == "LOW"
	risk < 40

	insurance_status == "absent"

	insurance_status == "unreported"
