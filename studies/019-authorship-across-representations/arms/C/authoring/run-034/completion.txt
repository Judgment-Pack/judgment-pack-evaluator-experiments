package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

decision := result if {
	evidence := object.get(input, "evidence", {})
	financial_status := object.get(evidence, "financial-evidence", "unreported")
	vendor := object.get(input, "vendor", {})
	result := decide(financial_status, vendor, evidence)
}

decide(financial_status, _, _) := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_status == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_status == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	financial_status == "present"
	object.get(input.vendor, "sanctionsStatus", "UNKNOWN") == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	financial_status == "present"
	object.get(input.vendor, "sanctionsStatus", "UNKNOWN") == "UNKNOWN"
} else := result if {
	financial_status == "present"
	object.get(input.vendor, "sanctionsStatus", "UNKNOWN") == "CLEAR"
	vendor := object.get(input, "vendor", {})
	evidence := object.get(input, "evidence", {})
	result := clear_result(vendor, evidence)
}

clear_result(vendor, evidence) := result if {
	outcomes := {
		outcome |
		some risk in risk_values(vendor)
		some spend in spend_values(vendor)
		some country in country_values(vendor)
		outcome := readable_outcome(vendor, evidence, risk, spend, country)
	}
	count(outcomes) == 1
	some result in outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

risk_values(vendor) := [vendor.riskScore] if {
	"riskScore" in vendor
} else := numbers.range(0, 100)

spend_values(vendor) := [vendor.requestedSpend] if {
	"requestedSpend" in vendor
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

country_values(vendor) := [vendor.countryRisk] if {
	"countryRisk" in vendor
} else := ["LOW", "MEDIUM", "HIGH"]

readable_outcome(_, _, _, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	risk := object.get(input.vendor, "riskScore", 0)
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	risk := object.get(input.vendor, "riskScore", 0)
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk := object.get(input.vendor, "riskScore", 0)
	risk < 40
	spend <= 500000
} else := result if {
	country == "LOW"
	risk := object.get(input.vendor, "riskScore", 0)
	risk < 40
	spend > 500000
	spend <= 2000000
	result := low_country_insured_result
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk := object.get(input.vendor, "riskScore", 0)
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "MEDIUM"
	risk := object.get(input.vendor, "riskScore", 0)
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []}

low_country_insured_result := {"disposition": "approve", "reasons": []} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]}
