package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_evidence == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_evidence == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	input.vendor.sanctionsStatus == "CLEAR"
	result := clear_result
}

financial_evidence := object.get(
	object.get(input, "evidence", {}),
	"financial-evidence",
	"unreported",
)

insurance_evidence := object.get(
	object.get(input, "evidence", {}),
	"insurance-certificate",
	"unreported",
)

critical_supplier := object.get(input.vendor, "criticalSupplier", "no")

prior_enforcement := object.get(input.vendor, "priorEnforcement", "no")

new_vendor := object.get(input.vendor, "newVendor", "no")

clear_result := outcome if {
	outcomes := {
		candidate_outcome(risk, spend, country) |
		some risk in risk_values
		some spend in spend_values
		some country in country_values
	}
	count(outcomes) == 1
	some outcome in outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

risk_values contains risk if {
	"riskScore" in input.vendor
	risk := input.vendor.riskScore
}

risk_values contains risk if {
	not "riskScore" in input.vendor
	some risk in numbers.range(0, 100)
}

spend_values contains spend if {
	"requestedSpend" in input.vendor
	spend := input.vendor.requestedSpend
}

spend_values contains spend if {
	not "requestedSpend" in input.vendor
	some spend in [
		0,
		100000,
		100000.01,
		500000,
		500000.01,
		2000000,
		2000000.01,
		10000000,
	]
}

country_values contains country if {
	"countryRisk" in input.vendor
	country := input.vendor.countryRisk
}

country_values contains country if {
	not "countryRisk" in input.vendor
	some country in ["LOW", "MEDIUM", "HIGH"]
}

candidate_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	critical_supplier == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	prior_enforcement == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	prior_enforcement != "yes"
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	prior_enforcement != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_evidence == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	prior_enforcement != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_evidence == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	prior_enforcement != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_evidence == "unreported"
} else := {"disposition": "approve", "reasons": []} if {
	prior_enforcement != "yes"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	new_vendor != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	prior_enforcement != "yes"
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []}
