package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

vendor := object.get(input, "vendor", {})
evidence := object.get(input, "evidence", {})

financial_status := object.get(evidence, "financial-evidence", "unreported")
insurance_status := object.get(evidence, "insurance-certificate", "unreported")

critical_supplier := object.get(vendor, "criticalSupplier", "no")
prior_enforcement := object.get(vendor, "priorEnforcement", "no")
new_vendor := object.get(vendor, "newVendor", "no")

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_status == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_status == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	vendor.sanctionsStatus == "CLEAR"
	result := clear_decision
}

clear_decision := result if {
	count(clear_outcomes) == 1
	some result in clear_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

clear_outcomes contains result if {
	some risk in possible_risk_scores
	some spend in possible_requested_spends
	some country in possible_country_risks
	result := readable_outcome(risk, spend, country)
}

possible_risk_scores := [vendor.riskScore] if {
	"riskScore" in vendor
} else := [0, 39, 40, 69, 70, 89, 90]

possible_requested_spends := [vendor.requestedSpend] if {
	"requestedSpend" in vendor
} else := [0, 100000, 500000, 500000.01, 2000000, 2000000.01]

possible_country_risks := [vendor.countryRisk] if {
	"countryRisk" in vendor
} else := ["LOW", "MEDIUM", "HIGH"]

readable_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	critical_supplier == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	prior_enforcement == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "unreported"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	new_vendor != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []}
