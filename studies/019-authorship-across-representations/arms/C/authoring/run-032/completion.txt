package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

decision := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	vendor := object.get(input, "vendor", {})
	evidence := object.get(input, "evidence", {})
	not "financial-evidence" in evidence
} else := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "absent"
} else := {"disposition": "reject", "reasons": []} if {
	vendor := object.get(input, "vendor", {})
	vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	vendor := object.get(input, "vendor", {})
	vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	vendor := object.get(input, "vendor", {})
	vendor.sanctionsStatus == "CLEAR"

	outcomes := {
		outcome |
		some risk in candidate_risks(vendor)
		some spend in candidate_spends(vendor)
		some country in candidate_countries(vendor)
		outcome := readable_outcome(vendor, risk, spend, country)
	}

	count(outcomes) == 1
	some only in outcomes
	result := outcome_result(only)
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	vendor := object.get(input, "vendor", {})
	vendor.sanctionsStatus == "CLEAR"

	outcomes := {
		outcome |
		some risk in candidate_risks(vendor)
		some spend in candidate_spends(vendor)
		some country in candidate_countries(vendor)
		outcome := readable_outcome(vendor, risk, spend, country)
	}

	count(outcomes) > 1
}

candidate_risks(vendor) := [vendor.riskScore] if {
	"riskScore" in vendor
} else := [0, 39, 40, 69, 70, 89, 90]

candidate_spends(vendor) := [vendor.requestedSpend] if {
	"requestedSpend" in vendor
} else := [0, 100000, 100000.01, 500000, 500000.01, 2000000, 2000000.01]

candidate_countries(vendor) := [vendor.countryRisk] if {
	"countryRisk" in vendor
} else := ["LOW", "MEDIUM", "HIGH"]

readable_outcome(vendor, risk, spend, country) := "exception-escalation" if {
	country == "HIGH"
	spend > 2000000
} else := "review" if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := "reject" if {
	risk >= 90
} else := "reject" if {
	country == "HIGH"
	risk >= 70
} else := "reject" if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := "approve" if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := d6b_outcome if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	d6b_outcome := insurance_outcome
} else := "approve" if {
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(vendor, "newVendor", "no") != "yes"
} else := "approve" if {
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := "review"

insurance_outcome := "unknown" if {
	evidence := object.get(input, "evidence", {})
	not "insurance-certificate" in evidence
} else := "approve" if {
	input.evidence["insurance-certificate"] == "present"
} else := "enhanced-review"

outcome_result("approve") := {"disposition": "approve", "reasons": []}
outcome_result("review") := {"disposition": "review", "reasons": []}
outcome_result("enhanced-review") := {"disposition": "enhanced-review", "reasons": []}
outcome_result("reject") := {"disposition": "reject", "reasons": []}
outcome_result("unknown") := {"disposition": "unresolved", "reasons": ["unknown"]}
outcome_result("exception-escalation") := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
}
