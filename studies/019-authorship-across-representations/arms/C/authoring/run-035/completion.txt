package study_test

import data.study

financial_evidence := {
	"financial-evidence": "present",
}

base_vendor := {
	"riskScore": 20,
	"requestedSpend": 1000,
	"sanctionsStatus": "CLEAR",
	"countryRisk": "LOW",
	"newVendor": "no",
	"criticalSupplier": "no",
	"priorEnforcement": "no",
}

test_missing_financial_evidence_precedes_rejection if {
	actual := study.decision with input as {
		"vendor": object.union(base_vendor, {"sanctionsStatus": "MATCH"}),
		"evidence": {"financial-evidence": "absent"},
	}

	actual == {
		"disposition": "unresolved",
		"reasons": ["missing-required-evidence"],
	}
}

test_unreported_financial_evidence_is_unknown if {
	actual := study.decision with input as {
		"vendor": base_vendor,
		"evidence": {},
	}

	actual == {
		"disposition": "unresolved",
		"reasons": ["unknown"],
	}
}

test_sanctions_match_rejects if {
	actual := study.decision with input as {
		"vendor": object.union(base_vendor, {
			"sanctionsStatus": "MATCH",
			"criticalSupplier": "yes",
		}),
		"evidence": financial_evidence,
	}

	actual == {"disposition": "reject", "reasons": []}
}
