package study

# Representative values partition each unreadable numeric input at every
# threshold used by the policy. U1 therefore needs only these assignments.
risk_values(vendor) := {risk} if {
	risk := object.get(vendor, "riskScore", -1)
	risk >= 0
} else := {0, 39, 40, 69, 70, 89, 90, 100} if {
	object.get(vendor, "riskScore", -1) == -1
}

spend_values(vendor) := {spend} if {
	spend := object.get(vendor, "requestedSpend", -1)
	spend >= 0
} else := {
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
} if {
	object.get(vendor, "requestedSpend", -1) == -1
}

country_values(vendor) := {country} if {
	country := object.get(vendor, "countryRisk", "")
	country != ""
} else := {"LOW", "MEDIUM", "HIGH"} if {
	object.get(vendor, "countryRisk", "") == ""
}

# P1: financial evidence is absent.
outcome(vendor, evidence) := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	object.get(evidence, "financial-evidence", "unreported") == "absent"
}

# P1: financial-evidence availability is unreported.
else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	object.get(evidence, "financial-evidence", "unreported") == "unreported"
}

# D1, which stands despite O2 when sanctions match.
else := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "MATCH"
}

# D2, which stands despite O2 when screening returned UNKNOWN.
else := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	object.get(vendor, "sanctionsStatus", "") == "UNKNOWN"
}

# O3 has priority over O2 and all determination clauses.
else := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "HIGH"
	object.get(vendor, "requestedSpend", -1) > 2000000
}

# O2 applies only to CLEAR results and is reached only when O3 did not apply.
else := {
	"disposition": "review",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "criticalSupplier", "no") == "yes"
}

# D3.
else := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "riskScore", -1) >= 90
}

# D4.
else := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "HIGH"
	object.get(vendor, "riskScore", -1) >= 70
}

# D5. Unreported prior-enforcement status is treated as no.
else := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") == "yes"
}

# D6a.
else := {
	"disposition": "approve",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "LOW"
	object.get(vendor, "riskScore", -1) < 40
	object.get(vendor, "requestedSpend", -1) <= 500000
}

# D6b: insurance is present.
else := {
	"disposition": "approve",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "LOW"
	object.get(vendor, "riskScore", -1) < 40
	object.get(vendor, "requestedSpend", -1) > 500000
	object.get(vendor, "requestedSpend", -1) <= 2000000
	object.get(evidence, "insurance-certificate", "unreported") == "present"
}

# D6b: insurance is absent.
else := {
	"disposition": "enhanced-review",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "LOW"
	object.get(vendor, "riskScore", -1) < 40
	object.get(vendor, "requestedSpend", -1) > 500000
	object.get(vendor, "requestedSpend", -1) <= 2000000
	object.get(evidence, "insurance-certificate", "unreported") == "absent"
}

# D6b: insurance availability is unreported.
else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "LOW"
	object.get(vendor, "riskScore", -1) < 40
	object.get(vendor, "requestedSpend", -1) > 500000
	object.get(vendor, "requestedSpend", -1) <= 2000000
	object.get(evidence, "insurance-certificate", "unreported") == "unreported"
}

# D6c, including O1's suspension for a reported new vendor.
else := {
	"disposition": "approve",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "LOW"
	object.get(vendor, "riskScore", -1) >= 40
	object.get(vendor, "riskScore", -1) < 70
	object.get(vendor, "requestedSpend", -1) <= 100000
	object.get(vendor, "newVendor", "no") != "yes"
}

# D7.
else := {
	"disposition": "approve",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
	object.get(vendor, "countryRisk", "") == "MEDIUM"
	object.get(vendor, "riskScore", -1) < 40
	object.get(vendor, "requestedSpend", -1) <= 100000
}

# D8.
else := {
	"disposition": "review",
	"reasons": [],
} if {
	object.get(vendor, "sanctionsStatus", "") == "CLEAR"
}

# U1 varies only unreadable risk, spend, and country inputs.
possible_outcomes contains result if {
	vendor := object.get(input, "vendor", {})
	evidence := object.get(input, "evidence", {})

	some risk in risk_values(vendor)
	some spend in spend_values(vendor)
	some country in country_values(vendor)

	readable_vendor := object.union(vendor, {
		"riskScore": risk,
		"requestedSpend": spend,
		"countryRisk": country,
	})

	result := outcome(readable_vendor, evidence)
}

decision := result if {
	count(possible_outcomes) == 1
	some result in possible_outcomes
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
}
