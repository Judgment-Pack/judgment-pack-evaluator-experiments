# P1: financial evidence is an absolute precondition.
decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],

	count(possible_outcomes) == 1
	some result in possible_outcomes

# U1 evaluates every outcome-relevant equivalence class for each unreadable
# numeric or country input. The written policy's comparisons partition the
# input domains at precisely these representative values.
risk_candidates := [input.vendor.riskScore] if {
	"riskScore" in input.vendor
} else := [0, 39, 40, 69, 70, 89, 90, 100]

spend_candidates := [input.vendor.requestedSpend] if {
	"requestedSpend" in input.vendor
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

country_candidates := [input.vendor.countryRisk] if {
	"countryRisk" in input.vendor
} else := ["LOW", "MEDIUM", "HIGH"]

possible_outcomes contains result if {
	some risk in risk_candidates
	some spend in spend_candidates
	some country in country_candidates

	vendor := object.union(input.vendor, {
		"riskScore": risk,
		"requestedSpend": spend,
		"countryRisk": country,
	})

	result := readable_outcome(vendor, input.evidence)
}

# Clause order: D1/D2, O3, O2, D3-D8 as modified by O1.
readable_outcome(vendor, evidence) := {
	"disposition": "reject",
	"reasons": [],

} if {

	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	vendor.sanctionsStatus == "CLEAR"
	vendor.riskScore >= 90
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	vendor.sanctionsStatus == "CLEAR"
	vendor.countryRisk == "HIGH"
	vendor.riskScore >= 70
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	vendor.sanctionsStatus == "CLEAR"
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {

	object.get(vendor, "priorEnforcement", "no") != "yes"
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend <= 500000
} else := d6b_outcome(evidence) if {

	object.get(vendor, "priorEnforcement", "no") != "yes"
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40

	vendor.requestedSpend <= 2000000

	"disposition": "approve",
