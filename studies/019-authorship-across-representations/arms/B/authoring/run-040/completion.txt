package study

# P1: financial evidence is an absolute precondition.
decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	input.evidence["financial-evidence"] == "absent"
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	not "financial-evidence" in input.evidence
}

# U1: evaluate every material assignment of unreadable numeric/country inputs.
# The representative values cover every policy interval and boundary.
decision := result if {
	input.evidence["financial-evidence"] == "present"
	outcomes := {
		outcome |
		some risk in risk_candidates
		some spend in spend_candidates
		some country in country_candidates
		outcome := readable_outcome(risk, spend, country)
	}
	count(outcomes) == 1
	some result in outcomes
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	input.evidence["financial-evidence"] == "present"
	outcomes := {
		outcome |
		some risk in risk_candidates
		some spend in spend_candidates
		some country in country_candidates
		outcome := readable_outcome(risk, spend, country)
	}
	count(outcomes) > 1
}

risk_candidates := [input.vendor.riskScore] if {
	"riskScore" in input.vendor
}

risk_candidates := [0, 39, 40, 69, 70, 89, 90, 100] if {
	not "riskScore" in input.vendor
}

spend_candidates := [input.vendor.requestedSpend] if {
	"requestedSpend" in input.vendor
}

spend_candidates := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
] if {
	not "requestedSpend" in input.vendor
}

country_candidates := [input.vendor.countryRisk] if {
	"countryRisk" in input.vendor
}

country_candidates := ["LOW", "MEDIUM", "HIGH"] if {
	not "countryRisk" in input.vendor
}

# O3 precedes O2 and every determination clause.
readable_outcome(_, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	spend > 2000000
}

# O2 applies only to CLEAR cases not captured by O3.
readable_outcome(_, spend, country) := {
	"disposition": "review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	input.vendor.criticalSupplier == "yes"
	not {
		country == "HIGH"
		spend > 2000000
	}
}

# D1 stands for MATCH, including for critical suppliers.
readable_outcome(_, _, _) := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "MATCH"
}

# D2: UNKNOWN is a present screening value for which no clause matches.
readable_outcome(_, _, _) := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
}

# The remaining ordered clauses apply to CLEAR, non-critical cases outside O3.
readable_outcome(risk, spend, country) := result if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "criticalSupplier", "no") != "yes"
	not {
		country == "HIGH"
		spend > 2000000
	}
	result := ordinary_clear_outcome(risk, spend, country)
}

# D3
ordinary_clear_outcome(risk, _, _) := {
	"disposition": "reject",
	"reasons": [],
} if {
	risk >= 90
}

# D4
ordinary_clear_outcome(risk, _, country) := {
	"disposition": "reject",
	"reasons": [],

# D5; an omitted prior-enforcement status is treated as no.
ordinary_clear_outcome(risk, _, country) := {

	"reasons": [],
} if {
	risk < 90
	not {
		country == "HIGH"

	not "insurance-certificate" in input.evidence
