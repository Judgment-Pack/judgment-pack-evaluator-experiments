package study

# P1: financial evidence is an absolute precondition.
decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "absent"
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "unreported"
}

# D1: sanctions matches reject regardless of the unreadable inputs.
decision := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "present"
	input.vendor.sanctionsStatus == "MATCH"
}

# D2: UNKNOWN is a present screening result for which no determination applies.
decision := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "present"
	input.vendor.sanctionsStatus == "UNKNOWN"
}

# CLEAR cases are evaluated over every relevant assignment to unreadable inputs.
# U1 issues an outcome only when all such assignments produce the same outcome.
decision := result if {
	object.get(input.evidence, "financial-evidence", "unreported") == "present"
	input.vendor.sanctionsStatus == "CLEAR"
	count(candidate_outcomes) == 1
	some result in candidate_outcomes
}

# If the U1 assignments differ, the unreadable input is material.
decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "present"
	input.vendor.sanctionsStatus == "CLEAR"
	count(candidate_outcomes) > 1
}

candidate_outcomes contains result if {
	some risk in possible_risk_scores
	some spend in possible_requested_spends
	some country in possible_country_risks
	result := clear_outcome(risk, spend, country)
}

# These representatives cover every outcome-changing interval and boundary.
possible_risk_scores contains risk if {
	risk := input.vendor.riskScore
}

possible_risk_scores contains risk if {
	not input.vendor.riskScore
	some risk in [0, 39, 40, 69, 70, 89, 90, 100]
}

possible_requested_spends contains spend if {
	spend := input.vendor.requestedSpend
}

possible_requested_spends contains spend if {
	not input.vendor.requestedSpend
	some spend in [
		0,
		100000,
		100000.01,
		500000,
		500000.01,
		2000000,
		2000000.01,
		10000000,
	]
}

possible_country_risks contains country if {
	country := input.vendor.countryRisk
}

possible_country_risks contains country if {
	not input.vendor.countryRisk
	some country in ["LOW", "MEDIUM", "HIGH"]
}

# O3 has first priority for CLEAR cases.
clear_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {
	"disposition": "review",
	"reasons": [],
} if {
	# O2 follows O3 and displaces all determination clauses.
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	# D3
	risk >= 90
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	# D4
	country == "HIGH"
	risk >= 70
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	# D5
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	# D6a
	country == "LOW"
	risk < 40
	spend <= 500000
} else := result if {
	# D6b, including its enhanced-review and unresolved limbs.
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	result := d6b_outcome
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	# D6c as modified by O1.
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	# D7
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {
	# D8
	"disposition": "review",
	"reasons": [],
}

d6b_outcome := {
	"disposition": "approve",
	"reasons": [],
} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "present"
} else := {
	"disposition": "enhanced-review",
	"reasons": [],
} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "absent"
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
}
