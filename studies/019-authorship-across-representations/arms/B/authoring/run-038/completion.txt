package study

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	input.vendor.sanctionsStatus == "CLEAR"
	results := possible_results
	count(results) == 1
	some result in results
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

possible_results := {
	result |
		some risk in risk_values
		some spend in spend_values
		some country in country_values
		result := readable_result(risk, spend, country)
}

risk_values := [input.vendor.riskScore] if {
	"riskScore" in object.keys(input.vendor)
} else := [0, 39, 40, 69, 70, 89, 90]

spend_values := [input.vendor.requestedSpend] if {
	"requestedSpend" in object.keys(input.vendor)
} else := [0, 100000, 100000.01, 500000, 500000.01, 2000000, 2000000.01]

country_values := [input.vendor.countryRisk] if {
	"countryRisk" in object.keys(input.vendor)
} else := ["LOW", "MEDIUM", "HIGH"]

readable_result(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := insurance_result if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_result := low_country_insurance_result
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []}

low_country_insurance_result := {"disposition": "approve", "reasons": []} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]}
