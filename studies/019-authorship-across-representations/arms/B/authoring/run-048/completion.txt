package study

determination(disposition) := {
	"disposition": disposition,
	"reasons": [],
}

unresolved(reasons) := {
	"disposition": "unresolved",
	"reasons": reasons,
}

financial_status := object.get(input.evidence, "financial-evidence", "unreported")

decision := unresolved(["missing-required-evidence"]) if {
	financial_status == "absent"
}

decision := unresolved(["unknown"]) if {
	financial_status == "unreported"
}

decision := determination("reject") if {
	financial_status == "present"
	input.vendor.sanctionsStatus == "MATCH"
}

decision := unresolved(["no-match"]) if {
	financial_status == "present"
	input.vendor.sanctionsStatus == "UNKNOWN"
}

decision := clear_decision if {
	financial_status == "present"
	input.vendor.sanctionsStatus == "CLEAR"
	candidate_outcomes := {
		outcome |
		some vendor in candidate_vendors
		outcome := readable_outcome(vendor)
	}
	count(candidate_outcomes) == 1
	some clear_decision in candidate_outcomes
}

decision := unresolved(["unknown"]) if {
	financial_status == "present"
	input.vendor.sanctionsStatus == "CLEAR"
	candidate_outcomes := {
		outcome |
		some vendor in candidate_vendors
		outcome := readable_outcome(vendor)
	}
	count(candidate_outcomes) > 1
}

risk_options := [input.vendor.riskScore] if {
	object.get(input.vendor, "riskScore", -1) >= 0
}

risk_options := numbers.range(0, 100) if {
	object.get(input.vendor, "riskScore", -1) == -1
}

spend_options := [input.vendor.requestedSpend] if {
	object.get(input.vendor, "requestedSpend", -1) >= 0
}

spend_options := [
	0,
	100000,
	500000,
	500000.01,
	2000000,
	2000000.01,
] if {
	object.get(input.vendor, "requestedSpend", -1) == -1
}

country_options := [input.vendor.countryRisk] if {
	object.get(input.vendor, "countryRisk", "") != ""
}

country_options := ["LOW", "MEDIUM", "HIGH"] if {
	object.get(input.vendor, "countryRisk", "") == ""
}

candidate_vendors contains vendor if {
	some risk in risk_options
	some spend in spend_options
	some country in country_options
	vendor := object.union(input.vendor, {
		"riskScore": risk,
		"requestedSpend": spend,
		"countryRisk": country,
	})
}

readable_outcome(vendor) := unresolved(["exception-escalation"]) if {
	vendor.countryRisk == "HIGH"
	vendor.requestedSpend > 2000000
} else := determination("review") if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := determination("reject") if {
	vendor.riskScore >= 90
} else := determination("reject") if {
	vendor.countryRisk == "HIGH"
	vendor.riskScore >= 70
} else := determination("reject") if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := determination("approve") if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend <= 500000
} else := determination("approve") if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	input.evidence["insurance-certificate"] == "present"
} else := determination("enhanced-review") if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	input.evidence["insurance-certificate"] == "absent"
} else := unresolved(["unknown"]) if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	object.get(input.evidence, "insurance-certificate", "unreported") == "unreported"
} else := determination("approve") if {
	vendor.countryRisk == "LOW"
	vendor.riskScore >= 40
	vendor.riskScore < 70
	vendor.requestedSpend <= 100000
	object.get(vendor, "newVendor", "no") != "yes"
} else := determination("approve") if {
	vendor.countryRisk == "MEDIUM"
	vendor.riskScore < 40
	vendor.requestedSpend <= 100000
} else := determination("review")
