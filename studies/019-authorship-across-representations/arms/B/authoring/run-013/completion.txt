```rego
package study

# Vendor Approval Policy implementation.
#
# Missing riskScore, requestedSpend, and countryRisk values are evaluated under
# U1 by considering every outcome-relevant readable assignment. If all such
# assignments produce the same outcome, that outcome is issued.

decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	object.get(input, "evidence", {})["financial-evidence"] == "absent"
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	evidence := object.get(input, "evidence", {})
	not "financial-evidence" in evidence
}

decision := result if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "present"

	results := candidate_results
	count(results) == 1
	some result in results
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "present"

	count(candidate_results) > 1
}

candidate_results := {
	result |
	some risk_score in risk_score_values
	some requested_spend in requested_spend_values
	some country_risk in country_risk_values
	result := readable_result(risk_score, requested_spend, country_risk)
}

risk_score_values := [input.vendor.riskScore] if {
	"riskScore" in input.vendor
}

risk_score_values := numbers.range(0, 100) if {
	not "riskScore" in input.vendor
}

requested_spend_values := [input.vendor.requestedSpend] if {
	"requestedSpend" in input.vendor
}

# These representatives cover every spend interval whose members have identical
# policy consequences.
requested_spend_values := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
] if {
	not "requestedSpend" in input.vendor
}

country_risk_values := [input.vendor.countryRisk] if {
	"countryRisk" in input.vendor
}

country_risk_values := ["LOW", "MEDIUM", "HIGH"] if {
	not "countryRisk" in input.vendor
}

# P1 has already been satisfied before readable_result is evaluated. The
# else-chain implements O3, O2, and D1-D8 in their required precedence order.
readable_result(risk_score, requested_spend, country_risk) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "HIGH"
	requested_spend > 2000000
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := {
	"disposition": "review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	risk_score >= 90
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "HIGH"
	risk_score >= 70
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "LOW"
	risk_score < 40
	requested_spend <= 500000
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "LOW"
	risk_score < 40
	requested_spend > 500000
	requested_spend <= 2000000
	object.get(input, "evidence", {})["insurance-certificate"] == "present"
} else := {
	"disposition": "enhanced-review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "LOW"
	risk_score < 40
	requested_spend > 500000
	requested_spend <= 2000000
	object.get(input, "evidence", {})["insurance-certificate"] == "absent"
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "LOW"
	risk_score < 40
	requested_spend > 500000
	requested_spend <= 2000000
	evidence := object.get(input, "evidence", {})
	not "insurance-certificate" in evidence
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "LOW"
	risk_score >= 40
	risk_score < 70
	requested_spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country_risk == "MEDIUM"
	risk_score < 40
	requested_spend <= 100000
} else := {
	"disposition": "review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
}
```
