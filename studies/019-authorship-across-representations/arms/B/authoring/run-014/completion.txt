	"disposition": "unresolved",

	value.vendor.countryRisk == "LOW"
	value.vendor.riskScore < 40
	value.vendor.requestedSpend > 500000

	not has_key(value.evidence, "insurance-certificate")

	"disposition": "approve",
	"reasons": [],

	# D6c, as limited by O1.

	# D8.
	"disposition": "review",
	"reasons": [],
