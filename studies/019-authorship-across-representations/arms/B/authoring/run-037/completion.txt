POLICY:
```rego
package study

determination(value) := {
	"disposition": value,
	"reasons": [],
}

unresolved(reason) := {
	"disposition": "unresolved",
	"reasons": [reason],
}

field_values(object, key, _) := [object[key]] if {
	key in object
}

field_values(object, key, defaults) := defaults if {
	not key in object
}

risk_values := field_values(
	input.vendor,
	"riskScore",
	[0, 39, 40, 69, 70, 89, 90, 100],
)

spend_values := field_values(
	input.vendor,
	"requestedSpend",
	[0, 100000, 100000.01, 500000, 500000.01, 2000000, 2000000.01, 10000000],
)

country_values := field_values(
	input.vendor,
	"countryRisk",
	["LOW", "MEDIUM", "HIGH"],
)

candidate_vendors contains candidate if {
	some risk in risk_values
	some spend in spend_values
	some country in country_values
	candidate := object.union(input.vendor, {
		"riskScore": risk,
		"requestedSpend": spend,
		"countryRisk": country,
	})
}

candidate_outcomes contains result if {
	some vendor in candidate_vendors
	result := outcome(vendor)
}

prior_enforcement(vendor) if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
}

new_vendor(vendor) if {
	object.get(vendor, "newVendor", "no") == "yes"
}

critical_supplier(vendor) if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
}

insurance_present if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "present"
}

insurance_absent if {
	object.get(input.evidence, "insurance-certificate", "unreported") == "absent"
}

# P1, followed by D1/D2, O3, O2, and D3-D8 in governing order.
outcome(vendor) := determination("reject") if {
	vendor.sanctionsStatus == "MATCH"
} else := unresolved("no-match") if {
	vendor.sanctionsStatus == "UNKNOWN"
} else := unresolved("exception-escalation") if {
	vendor.sanctionsStatus == "CLEAR"
	vendor.countryRisk == "HIGH"
	vendor.requestedSpend > 2000000
} else := determination("review") if {
	vendor.sanctionsStatus == "CLEAR"
	critical_supplier(vendor)
} else := determination("reject") if {
	vendor.sanctionsStatus == "CLEAR"
	vendor.riskScore >= 90
} else := determination("reject") if {
	vendor.sanctionsStatus == "CLEAR"
	vendor.countryRisk == "HIGH"
	vendor.riskScore >= 70
} else := determination("reject") if {
	vendor.sanctionsStatus == "CLEAR"
	prior_enforcement(vendor)
} else := determination("approve") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend <= 500000
} else := determination("approve") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	insurance_present
} else := determination("enhanced-review") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	insurance_absent
} else := unresolved("unknown") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	not "insurance-certificate" in input.evidence
} else := determination("approve") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "LOW"
	vendor.riskScore >= 40
	vendor.riskScore < 70
	vendor.requestedSpend <= 100000
	not new_vendor(vendor)
} else := determination("approve") if {
	vendor.sanctionsStatus == "CLEAR"
	not prior_enforcement(vendor)
	vendor.countryRisk == "MEDIUM"
	vendor.riskScore < 40
	vendor.requestedSpend <= 100000
} else := determination("review")

# P1: explicitly absent financial evidence.
decision := unresolved("missing-required-evidence") if {
	object.get(input.evidence, "financial-evidence", "unreported") == "absent"
}

# P1: unreported financial-evidence availability.
decision := unresolved("unknown") if {
	not "financial-evidence" in input.evidence
}

# Fully readable risk, spend, and country inputs.
decision := outcome(input.vendor) if {
	input.evidence["financial-evidence"] == "present"
	"riskScore" in input.vendor
	"requestedSpend" in input.vendor
	"countryRisk" in input.vendor
}

# U1: all readable assignments yield one identical outcome.
decision := result if {
	input.evidence["financial-evidence"] == "present"
	missing_evaluated_input
	count(candidate_outcomes) == 1
	some result in candidate_outcomes
}

# U1: readable assignments yield different outcomes.
decision := unresolved("unknown") if {
	input.evidence["financial-evidence"] == "present"
	missing_evaluated_input
	count(candidate_outcomes) > 1
}

missing_evaluated_input if {
	not "riskScore" in input.vendor
}

missing_evaluated_input if {
	not "requestedSpend" in input.vendor
}

missing_evaluated_input if {
	not "countryRisk" in input.vendor
}
```
