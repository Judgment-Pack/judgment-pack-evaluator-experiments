# P1: financial evidence is an absolute precondition.
decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "absent"
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	evidence := object.get(input, "evidence", {})
	not "financial-evidence" in evidence
}

# D1 and D2 apply after P1 and do not depend on unreadable vendor inputs.
decision := {
	"disposition": "reject",
	"reasons": [],
} if {
	financial_evidence_available
	input.vendor.sanctionsStatus == "MATCH"
}

decision := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	financial_evidence_available
	input.vendor.sanctionsStatus == "UNKNOWN"
}

# For CLEAR screening, U1 evaluates every outcome obtainable by assigning
# readable values to omitted risk, spend, and country inputs.
decision := outcome if {
	clear_case
	outcomes := possible_outcomes(input)
	count(outcomes) == 1
	some outcome in outcomes
}

decision := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	clear_case
	outcomes := possible_outcomes(input)
	count(outcomes) > 1
}

financial_evidence_available if {
	evidence := object.get(input, "evidence", {})
	evidence["financial-evidence"] == "present"
}

clear_case if {
	financial_evidence_available
	input.vendor.sanctionsStatus == "CLEAR"
}

possible_outcomes(request) := {
	outcome |
		some risk in possible_risks(request.vendor)
		some spend in possible_spends(request.vendor)
		some country in possible_countries(request.vendor)
		vendor := object.union(request.vendor, {
			"riskScore": risk,
			"requestedSpend": spend,
			"countryRisk": country,
		})
		outcome := readable_outcome(vendor, object.get(request, "evidence", {}))
}

possible_risks(vendor) := [risk] if {
	risk := vendor.riskScore
} else := [0, 39, 40, 69, 70, 89, 90, 100]

possible_spends(vendor) := [spend] if {
	spend := vendor.requestedSpend
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

possible_countries(vendor) := [country] if {
	country := vendor.countryRisk
} else := ["LOW", "MEDIUM", "HIGH"]

# O3 precedes O2 and every determination clause.
readable_outcome(vendor, evidence) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	vendor.countryRisk == "HIGH"
	vendor.requestedSpend > 2000000
} else := {
	"disposition": "review",
	"reasons": [],
} if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	vendor.riskScore >= 90
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	vendor.countryRisk == "HIGH"
	vendor.riskScore >= 70
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend <= 500000
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	evidence["insurance-certificate"] == "present"
} else := {
	"disposition": "enhanced-review",
	"reasons": [],
} if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	evidence["insurance-certificate"] == "absent"
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	vendor.countryRisk == "LOW"
	vendor.riskScore < 40
	vendor.requestedSpend > 500000
	vendor.requestedSpend <= 2000000
	not "insurance-certificate" in evidence
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	vendor.countryRisk == "LOW"
	vendor.riskScore >= 40
	vendor.riskScore < 70
	vendor.requestedSpend <= 100000
	object.get(vendor, "newVendor", "no") != "yes"

	"disposition": "approve",

} if {
	vendor.countryRisk == "MEDIUM"

package study_test

	"financial-evidence": "present",

	"riskScore": 20,
	"requestedSpend": 10000,
	"sanctionsStatus": "CLEAR",
	"countryRisk": "LOW",

	"priorEnforcement": "no",

test_p1_absent_financial_evidence_precedes_rejection if {
	actual := study.decision with input as {
		"vendor": object.union(base_vendor, {
			"riskScore": 95,
			"sanctionsStatus": "MATCH",
		}),
		"evidence": {"financial-evidence": "absent"},
	}
