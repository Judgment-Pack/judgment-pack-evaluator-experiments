package study

# The finite representative values cover every interval and boundary at which
# varying an unreadable input can change the policy outcome.
risk_representatives := [0, 39, 40, 69, 70, 89, 90, 100]

spend_representatives := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

country_representatives := ["LOW", "MEDIUM", "HIGH"]

risk_values := [input.vendor.riskScore] if {
	"riskScore" in object.keys(input.vendor)
} else := risk_representatives

spend_values := [input.vendor.requestedSpend] if {
	"requestedSpend" in object.keys(input.vendor)
} else := spend_representatives

country_values := [input.vendor.countryRisk] if {
	"countryRisk" in object.keys(input.vendor)
} else := country_representatives

reported_vendor := object.get(input, "vendor", {})
reported_evidence := object.get(input, "evidence", {})

financial_status := object.get(reported_evidence, "financial-evidence", "unreported")
insurance_status := object.get(reported_evidence, "insurance-certificate", "unreported")

critical_status := object.get(reported_vendor, "criticalSupplier", "no")
new_vendor_status := object.get(reported_vendor, "newVendor", "no")
prior_enforcement_status := object.get(reported_vendor, "priorEnforcement", "no")

# O3: large exposure in a high-risk country.
readable_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	spend > 2000000
} else := {
	"disposition": "review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	critical_status == "yes"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	risk >= 90
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	risk >= 70
} else := {
	"disposition": "reject",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status == "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "present"
} else := {
	"disposition": "enhanced-review",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "absent"
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "unreported"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	new_vendor_status != "yes"
} else := {
	"disposition": "approve",
	"reasons": [],
} if {
	input.vendor.sanctionsStatus == "CLEAR"
	prior_enforcement_status != "yes"
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {
	"disposition": "review",
	"reasons": [],
}

possible_outcomes := {
	outcome |
	some risk in risk_values
	some spend in spend_values
	some country in country_values
	outcome := readable_outcome(risk, spend, country)
}

# P1 always applies before every override and determination.
decision := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
} if {
	financial_status == "absent"
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	financial_status == "unreported"
} else := outcome if {
	financial_status == "present"
	count(possible_outcomes) == 1
	some outcome in possible_outcomes
} else := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
} if {
	financial_status == "present"
	count(possible_outcomes) != 1
}
