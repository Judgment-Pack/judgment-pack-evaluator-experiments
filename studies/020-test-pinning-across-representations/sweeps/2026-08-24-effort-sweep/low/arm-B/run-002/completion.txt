package study

has_vendor_field(name) if {
	name in object.keys(object.get(input, "vendor", {}))
}

has_evidence_field(name) if {
	name in object.keys(object.get(input, "evidence", {}))
}

financial_evidence_available if {
	object.get(object.get(input, "evidence", {}), "financial-evidence", "") == "present"
}

financial_evidence_absent if {
	object.get(object.get(input, "evidence", {}), "financial-evidence", "") == "absent"
}

risk_values := [input.vendor.riskScore] if {
	has_vendor_field("riskScore")
}

risk_values := [0, 39, 40, 69, 70, 89, 90, 100] if {
	not has_vendor_field("riskScore")
}

spend_values := [input.vendor.requestedSpend] if {
	has_vendor_field("requestedSpend")
}

spend_values := [0, 100000, 100000.01, 500000, 500000.01, 2000000, 2000000.01, 10000000] if {
	not has_vendor_field("requestedSpend")
}

country_values := [input.vendor.countryRisk] if {
	has_vendor_field("countryRisk")
}

country_values := ["LOW", "MEDIUM", "HIGH"] if {
	not has_vendor_field("countryRisk")
}

outcome(risk, spend, country) := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := {"disposition": "unresolved", "reasons": ["exception-escalation"]} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	object.get(input.evidence, "insurance-certificate", "") == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	object.get(input.evidence, "insurance-certificate", "") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	not has_evidence_field("insurance-certificate")
} else := {"disposition": "approve", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
	object.get(input.vendor, "priorEnforcement", "no") != "yes"
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []} if {
	input.vendor.sanctionsStatus == "CLEAR"
}

possible_outcomes := {result |
	some risk in risk_values
	some spend in spend_values
	some country in country_values
	result := outcome(risk, spend, country)
}

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_evidence_absent
}

decision := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	not has_evidence_field("financial-evidence")
}

decision := result if {
	financial_evidence_available
	count(possible_outcomes) == 1
	some result in possible_outcomes
}

decision := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_evidence_available
	count(possible_outcomes) != 1
}
