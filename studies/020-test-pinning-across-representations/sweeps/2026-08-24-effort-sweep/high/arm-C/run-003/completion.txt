package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

vendor := object.get(input, "vendor", {})

evidence := object.get(input, "evidence", {})

financial_evidence := object.get(evidence, "financial-evidence", "unreported")

insurance_certificate := object.get(evidence, "insurance-certificate", "unreported")

sanctions_status := object.get(vendor, "sanctionsStatus", "unreported")

critical_supplier if {
	object.get(vendor, "criticalSupplier", "no") == "yes"
}

prior_enforcement if {
	object.get(vendor, "priorEnforcement", "no") == "yes"
}

new_vendor if {
	object.get(vendor, "newVendor", "no") == "yes"
}

risk_values := [risk] if {
	risk := object.get(vendor, "riskScore", -1)
	risk != -1
} else := [0, 39, 40, 69, 70, 89, 90, 100]

spend_values := [spend] if {
	spend := object.get(vendor, "requestedSpend", -1)
	spend != -1
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
	10000000,
]

country_values := [country] if {
	country := object.get(vendor, "countryRisk", "unreadable")
	country != "unreadable"
} else := ["LOW", "MEDIUM", "HIGH"]

determination(disposition) := {
	"disposition": disposition,
	"reasons": [],
}

unresolved(reason) := {
	"disposition": "unresolved",
	"reasons": [reason],
}

readable_outcome(risk, spend, country) := unresolved("missing-required-evidence") if {
	financial_evidence == "absent"
} else := unresolved("unknown") if {
	financial_evidence == "unreported"
} else := determination("reject") if {
	sanctions_status == "MATCH"
} else := unresolved("no-match") if {
	sanctions_status == "UNKNOWN"
} else := unresolved("exception-escalation") if {
	sanctions_status == "CLEAR"
	country == "HIGH"
	spend > 2000000
} else := determination("review") if {
	sanctions_status == "CLEAR"
	critical_supplier
} else := determination("reject") if {
	sanctions_status == "CLEAR"
	risk >= 90
} else := determination("reject") if {
	sanctions_status == "CLEAR"
	country == "HIGH"
	risk >= 70
} else := determination("reject") if {
	sanctions_status == "CLEAR"
	prior_enforcement
} else := determination("approve") if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend <= 500000
} else := determination("approve") if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "present"
} else := determination("enhanced-review") if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "absent"
} else := unresolved("unknown") if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "unreported"
} else := determination("approve") if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	not new_vendor
} else := determination("approve") if {
	sanctions_status == "CLEAR"
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := determination("review") if {
	sanctions_status == "CLEAR"
}

candidate_outcomes contains outcome if {
	some risk in risk_values
	some spend in spend_values
	some country in country_values
	outcome := readable_outcome(risk, spend, country)
}

decision := outcome if {
	count(candidate_outcomes) == 1
	some outcome in candidate_outcomes
} else := unresolved("unknown") if {
	count(candidate_outcomes) > 1
}
