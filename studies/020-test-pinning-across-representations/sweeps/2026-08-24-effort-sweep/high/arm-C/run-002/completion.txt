POLICY:
```rego
package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

vendor := object.get(input, "vendor", {})

evidence := object.get(input, "evidence", {})

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	evidence["financial-evidence"] == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	not "financial-evidence" in object.keys(evidence)
} else := {"disposition": "reject", "reasons": []} if {
	vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	vendor.sanctionsStatus == "UNKNOWN"
} else := clear_decision if {
	vendor.sanctionsStatus == "CLEAR"
}

risk_values := [vendor.riskScore] if {
	"riskScore" in object.keys(vendor)
} else := [0, 40, 70, 90]

spend_values := [vendor.requestedSpend] if {
	"requestedSpend" in object.keys(vendor)
} else := [0, 100000.01, 500000.01, 2000000.01]

country_values := [vendor.countryRisk] if {
	"countryRisk" in object.keys(vendor)
} else := ["LOW", "MEDIUM", "HIGH"]

candidate_outcomes := {
	outcome |
	some risk in risk_values
	some spend in spend_values
	some country in country_values
	outcome := candidate_outcome(risk, spend, country)
}

clear_decision := outcome if {
	count(candidate_outcomes) == 1
	some outcome in candidate_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]}

candidate_values_valid(risk, spend, country) if {
	risk >= 0
	risk <= 100
	spend >= 0
	spend <= 10000000
	country in {"LOW", "MEDIUM", "HIGH"}
}

candidate_outcome(risk, spend, country) := {"disposition": "unresolved", "reasons": ["exception-escalation"]} if {
	candidate_values_valid(risk, spend, country)
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	object.get(vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	object.get(vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	evidence["insurance-certificate"] == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	evidence["insurance-certificate"] == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	candidate_values_valid(risk, spend, country)
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	not "insurance-certificate" in object.keys(evidence)
} else := {"disposition": "approve", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []} if {
	candidate_values_valid(risk, spend, country)
}
```
