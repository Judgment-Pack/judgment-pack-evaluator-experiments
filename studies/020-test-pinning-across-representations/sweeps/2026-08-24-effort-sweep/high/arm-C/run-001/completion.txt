POLICY:
```rego
package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

vendor := object.get(input, "vendor", {})
evidence := object.get(input, "evidence", {})

financial_state := object.get(evidence, "financial-evidence", "unreported")
insurance_state := object.get(evidence, "insurance-certificate", "unreported")
sanctions_state := object.get(vendor, "sanctionsStatus", "unreported")
critical_supplier := object.get(vendor, "criticalSupplier", "no")
new_vendor := object.get(vendor, "newVendor", "no")
prior_enforcement := object.get(vendor, "priorEnforcement", "no")

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	financial_state == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	financial_state != "absent"
	financial_state != "present"
} else := {"disposition": "reject", "reasons": []} if {
	sanctions_state == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	sanctions_state == "UNKNOWN"
} else := clear_decision if {
	sanctions_state == "CLEAR"
}

risk_value := object.get(vendor, "riskScore", -1)

risk_candidates := [risk_value] if {
	risk_value != -1
} else := [0, 39, 40, 69, 70, 89, 90, 100] if true

spend_value := object.get(vendor, "requestedSpend", -1)

spend_candidates := [spend_value] if {
	spend_value != -1
} else := [
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
] if true

country_value := object.get(vendor, "countryRisk", "__missing__")

country_candidates := [country_value] if {
	country_value != "__missing__"
} else := ["LOW", "MEDIUM", "HIGH"] if true

possible_outcomes := {outcome |
	some risk in risk_candidates
	some spend in spend_candidates
	some country in country_candidates
	outcome := readable_outcome(risk, spend, country)
}

clear_decision := outcome if {
	count(possible_outcomes) == 1
	some outcome in possible_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if true

readable_outcome(risk, spend, country) := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	critical_supplier == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	prior_enforcement == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_state == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_state == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_state == "unreported"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	new_vendor != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []} if true
```
