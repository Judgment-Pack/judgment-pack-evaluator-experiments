package study

vendor_input := object.get(input, "vendor", {})

evidence_input := object.get(input, "evidence", {})

financial_status := object.get(evidence_input, "financial-evidence", "unreported")

insurance_status := object.get(evidence_input, "insurance-certificate", "unreported")

critical_supplier_status := object.get(vendor_input, "criticalSupplier", "no")

prior_enforcement_status := object.get(vendor_input, "priorEnforcement", "no")

new_vendor_status := object.get(vendor_input, "newVendor", "no")

determination(disposition) := {
	"disposition": disposition,
	"reasons": [],
}

unresolved(ground) := {
	"disposition": "unresolved",
	"reasons": [ground],
}

# P1: Financial evidence is an absolute precondition.
decision := unresolved("missing-required-evidence") if {
	financial_status == "absent"
}

decision := unresolved("unknown") if {
	financial_status == "unreported"
}

# D1 and D2 apply before consideration of the inputs governed by U1.
decision := determination("reject") if {
	financial_status == "present"
	vendor_input.sanctionsStatus == "MATCH"
}

decision := unresolved("no-match") if {
	financial_status == "present"
	vendor_input.sanctionsStatus == "UNKNOWN"
}

decision := clear_decision if {
	financial_status == "present"
	vendor_input.sanctionsStatus == "CLEAR"
}

# U1: Present values are used directly. An omitted risk score ranges over every
# readable integer allowed by the input definition.
risk_candidates contains risk if {
	"riskScore" in vendor_input
	risk := vendor_input.riskScore
}

risk_candidates contains risk if {
	not "riskScore" in vendor_input
	some risk in numbers.range(0, 100)
}

# These representatives cover every spend interval distinguished by the policy:
# <= 100,000; > 100,000 and <= 500,000; > 500,000 and <= 2,000,000;
# and > 2,000,000.
representative_spends := [
	0,
	100000.01,
	500000.01,
	2000000.01,
]

spend_candidates contains spend if {
	"requestedSpend" in vendor_input
	spend := vendor_input.requestedSpend
}

spend_candidates contains spend if {
	not "requestedSpend" in vendor_input
	some spend in representative_spends
}

country_candidates contains country if {
	"countryRisk" in vendor_input
	country := vendor_input.countryRisk
}

country_candidates contains country if {
	not "countryRisk" in vendor_input
	some country in ["LOW", "MEDIUM", "HIGH"]
}

clear_outcomes := {outcome |
	some risk in risk_candidates
	some spend in spend_candidates
	some country in country_candidates
	outcome := outcome_for(risk, spend, country)
}

# U1 issues an outcome only when every readable completion yields that same
# outcome object.
clear_decision := outcome if {
	count(clear_outcomes) == 1
	some outcome in clear_outcomes
}

clear_decision := unresolved("unknown") if {
	count(clear_outcomes) > 1
}

# O3 has first priority after P1 for CLEAR screenings.
outcome_for(risk, spend, country) := unresolved("exception-escalation") if {
	country == "HIGH"
	spend > 2000000
}

# O2 precedes and displaces every determination clause for remaining CLEAR cases.
else := determination("review") if {
	critical_supplier_status == "yes"
}

# D3.
else := determination("reject") if {
	risk >= 90
}

# D4.
else := determination("reject") if {
	country == "HIGH"
	risk >= 70
}

# D5.
else := determination("reject") if {
	prior_enforcement_status == "yes"
}

# D6a.
else := determination("approve") if {
	country == "LOW"
	risk < 40
	spend <= 500000
}

# D6b, available insurance.
else := determination("approve") if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "present"
}

# D6b, absent insurance.
else := determination("enhanced-review") if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "absent"
}

# D6b, unreported insurance availability.
else := unresolved("unknown") if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_status == "unreported"
}

# D6c as modified by O1.
else := determination("approve") if {
	country == "LOW"

	risk < 70
	spend <= 100000
	new_vendor_status != "yes"
}
