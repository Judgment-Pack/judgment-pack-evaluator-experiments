package study

vendor := object.get(input, "vendor", {})
evidence := object.get(input, "evidence", {})

financial_evidence := object.get(evidence, "financial-evidence", "unreported")
insurance_certificate := object.get(evidence, "insurance-certificate", "unreported")

sanctions_status := object.get(vendor, "sanctionsStatus", "UNKNOWN")
new_vendor := object.get(vendor, "newVendor", "no")
critical_supplier := object.get(vendor, "criticalSupplier", "no")
prior_enforcement := object.get(vendor, "priorEnforcement", "no")

approve_result := {"disposition": "approve", "reasons": []}
review_result := {"disposition": "review", "reasons": []}
enhanced_review_result := {"disposition": "enhanced-review", "reasons": []}
reject_result := {"disposition": "reject", "reasons": []}

missing_evidence_result := {
	"disposition": "unresolved",
	"reasons": ["missing-required-evidence"],
}

unknown_result := {
	"disposition": "unresolved",
	"reasons": ["unknown"],
}

no_match_result := {
	"disposition": "unresolved",
	"reasons": ["no-match"],
}

escalation_result := {
	"disposition": "unresolved",
	"reasons": ["exception-escalation"],
}

# These finite representative sets cover every outcome-relevant interval in
# the permitted numeric domains. They are used only for U1's quantification
# over omitted risk, spend, and country inputs.
risk_candidates := {vendor.riskScore} if {
	"riskScore" in object.keys(vendor)
} else := {0, 39, 40, 69, 70, 89, 90}

spend_candidates := {vendor.requestedSpend} if {
	"requestedSpend" in object.keys(vendor)
} else := {
	0,
	100000,
	100000.01,
	500000,
	500000.01,
	2000000,
	2000000.01,
}

country_candidates := {vendor.countryRisk} if {
	"countryRisk" in object.keys(vendor)
} else := {"LOW", "MEDIUM", "HIGH"}

# P1: absent evidence and unreported availability take precedence over all
# other clauses.
decision := missing_evidence_result if {
	financial_evidence == "absent"
}

decision := unknown_result if {
	financial_evidence == "unreported"
}

# U1: issue an outcome only when all readable assignments to omitted numeric
# or country inputs produce the identical outcome.
decision := outcome if {
	financial_evidence == "present"
	possible_outcomes := {
		candidate_outcome |
		some risk in risk_candidates
		some spend in spend_candidates
		some country in country_candidates
		candidate_outcome := outcome_for(risk, spend, country)
	}
	count(possible_outcomes) == 1
	some outcome in possible_outcomes
}

decision := unknown_result if {
	financial_evidence == "present"
	possible_outcomes := {
		candidate_outcome |
		some risk in risk_candidates
		some spend in spend_candidates
		some country in country_candidates
		candidate_outcome := outcome_for(risk, spend, country)
	}
	count(possible_outcomes) > 1
}

# O3: large exposure in a high-risk country.
outcome_for(_, spend, country) := escalation_result if {
	sanctions_status == "CLEAR"
	country == "HIGH"
	spend > 2000000
}

# O2: critical-supplier override, after O3 and only for CLEAR screening.
else := review_result if {
	sanctions_status == "CLEAR"
	critical_supplier == "yes"
}

# D1: sanctions match.
else := reject_result if {
	sanctions_status == "MATCH"
}

# D2, including an omitted sanctionsStatus.
else := no_match_result if {
	sanctions_status == "UNKNOWN"
}

# D3: critical risk.
else := reject_result if {
	sanctions_status == "CLEAR"
	risk >= 90
}

# D4: elevated risk in a high-risk country.
else := reject_result if {
	sanctions_status == "CLEAR"
	country == "HIGH"
	risk >= 70
}

# D5: recorded prior enforcement action.
else := reject_result if {
	sanctions_status == "CLEAR"
	prior_enforcement == "yes"
}

# D6a: low risk, low-country risk, spend through $500,000.
else := approve_result if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend <= 500000
}

# D6b: insurance certificate available.
else := approve_result if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "present"
}

# D6b: insurance certificate expressly absent.
else := enhanced_review_result if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "absent"
}

# D6b: insurance availability unreported.
else := unknown_result if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	insurance_certificate == "unreported"
}

# D6c, as modified by O1.
else := approve_result if {
	sanctions_status == "CLEAR"
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	new_vendor != "yes"
}

# D7: medium-country approval.
else := approve_result if {
	sanctions_status == "CLEAR"
	country == "MEDIUM"
	risk < 40
	spend <= 100000
}

# D8: every remaining CLEAR request.
else := review_result if {
	sanctions_status == "CLEAR"
}
