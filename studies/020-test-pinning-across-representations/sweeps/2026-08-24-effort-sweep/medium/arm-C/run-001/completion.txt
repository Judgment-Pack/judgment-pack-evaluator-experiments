package study

default decision := {"disposition": "unresolved", "reasons": ["no-match"]}

decision := {"disposition": "unresolved", "reasons": ["missing-required-evidence"]} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	object.get(input.evidence, "financial-evidence", "unreported") == "unreported"
} else := {"disposition": "reject", "reasons": []} if {
	input.vendor.sanctionsStatus == "MATCH"
} else := {"disposition": "unresolved", "reasons": ["no-match"]} if {
	input.vendor.sanctionsStatus == "UNKNOWN"
} else := result if {
	input.vendor.sanctionsStatus == "CLEAR"
	result := clear_decision
}

clear_decision := result if {
	count(possible_outcomes) == 1
	some result in possible_outcomes
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	count(possible_outcomes) > 1
}

possible_outcomes := {result |
	some risk in possible_risks
	some spend in possible_spends
	some country in possible_countries
	result := outcome(risk, spend, country)
}

risk_readable if {
	object.get(input.vendor, "riskScore", -1) != -1
}

spend_readable if {
	object.get(input.vendor, "requestedSpend", -1) != -1
}

country_readable if {
	object.get(input.vendor, "countryRisk", "") != ""
}

possible_risks contains risk if {
	risk_readable
	risk := input.vendor.riskScore
}

possible_risks contains risk if {
	not risk_readable
	some risk in {0, 39, 40, 69, 70, 89, 90, 100}
}

possible_spends contains spend if {
	spend_readable
	spend := input.vendor.requestedSpend
}

possible_spends contains spend if {
	not spend_readable
	some spend in {
		0,
		100000,
		100000.01,
		500000,
		500000.01,
		2000000,
		2000000.01,
		10000000,
	}
}

possible_countries contains country if {
	country_readable
	country := input.vendor.countryRisk
}

possible_countries contains country if {
	not country_readable
	some country in {"LOW", "MEDIUM", "HIGH"}
}

outcome(risk, spend, country) := {"disposition": "unresolved", "reasons": ["exception-escalation"]} if {
	country == "HIGH"
	spend > 2000000
} else := {"disposition": "review", "reasons": []} if {
	object.get(input.vendor, "criticalSupplier", "no") == "yes"
} else := {"disposition": "reject", "reasons": []} if {
	risk >= 90
} else := {"disposition": "reject", "reasons": []} if {
	country == "HIGH"
	risk >= 70
} else := {"disposition": "reject", "reasons": []} if {
	object.get(input.vendor, "priorEnforcement", "no") == "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend <= 500000
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	object.get(input.evidence, "insurance-certificate", "unreported") == "present"
} else := {"disposition": "enhanced-review", "reasons": []} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	object.get(input.evidence, "insurance-certificate", "unreported") == "absent"
} else := {"disposition": "unresolved", "reasons": ["unknown"]} if {
	country == "LOW"
	risk < 40
	spend > 500000
	spend <= 2000000
	object.get(input.evidence, "insurance-certificate", "unreported") == "unreported"
} else := {"disposition": "approve", "reasons": []} if {
	country == "LOW"
	risk >= 40
	risk < 70
	spend <= 100000
	object.get(input.vendor, "newVendor", "no") != "yes"
} else := {"disposition": "approve", "reasons": []} if {
	country == "MEDIUM"
	risk < 40
	spend <= 100000
} else := {"disposition": "review", "reasons": []} if {
	true
}
